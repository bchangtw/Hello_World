  <script>

    const apiKey = "bd5e378503939ddaee76f12ad7a97608";

    let weatherChart = null;


    async function searchWeather() {

      const cityInput = document.getElementById("cityInput").value.trim();

      const msg = document.getElementById("msg");

      const hourly = document.getElementById("hourly");


      if (!cityInput) {

        msg.textContent = "請輸入城市名稱";

        return;

      }


      msg.textContent = "查詢中...";

      hourly.innerHTML = "";


      try {

        // 1. 中文或英文城市名稱 -> 經緯度

        const geoRes = await fetch(

          `https://api.openweathermap.org/geo/1.0/direct?q=${encodeURIComponent(cityInput)}&limit=1&appid=${apiKey}`

        );

        const geoData = await geoRes.json();


        if (!geoData || geoData.length === 0) {

          msg.textContent = "找不到城市，請重新輸入。";

          return;

        }


        const { lat, lon, name, country, state, local_names } = geoData[0];


        // 優先顯示中文名稱

        let displayName = name;

        if (local_names) {

          displayName =

            local_names.zh ||

            local_names["zh_tw"] ||

            local_names["zh_cn"] ||

            name;

        }


        // 2. 目前天氣

        const currentRes = await fetch(

          `https://api.openweathermap.org/data/2.5/weather?lat=${lat}&lon=${lon}&appid=${apiKey}&units=metric&lang=zh_tw`

        );

        const currentData = await currentRes.json();


        if (Number(currentData.cod) !== 200) {

          msg.textContent = "無法取得目前天氣資料。";

          return;

        }


        document.getElementById("city").textContent =

          state ? `${displayName}, ${state}, ${country}` : `${displayName}, ${country}`;

        document.getElementById("temp").textContent = `${Math.round(currentData.main.temp)}°C`;

        document.getElementById("desc").textContent = currentData.weather[0].description;

        document.getElementById("extraInfo").textContent =

          `體感 ${Math.round(currentData.main.feels_like)}°C ｜ 濕度 ${currentData.main.humidity}% ｜ 風速 ${currentData.wind.speed} m/s`;


        const mainIcon = document.getElementById("mainIcon");

        mainIcon.src = `https://openweathermap.org/img/wn/${currentData.weather[0].icon}@4x.png`;

        mainIcon.style.display = "block";


        // 3. 預報資料（每3小時一筆，4筆約12小時）

        const forecastRes = await fetch(

          `https://api.openweathermap.org/data/2.5/forecast?lat=${lat}&lon=${lon}&appid=${apiKey}&units=metric&lang=zh_tw`

        );

        const forecastData = await forecastRes.json();


        if (!forecastData.list || !forecastData.list.length) {

          msg.textContent = "無法取得預報資料。";

          return;

        }


        const list = forecastData.list.slice(0, 4);

        const labels = [];

        const temps = [];


        list.forEach(item => {

          const time = item.dt_txt.slice(11, 16);

          const temp = Math.round(item.main.temp);

          const desc = item.weather[0].description;

          const icon = item.weather[0].icon;


          labels.push(time);

          temps.push(temp);


          const box = document.createElement("div");

          box.className = "hour-box";

          box.innerHTML = `

            <div class="time">${time}</div>

            <img class="hour-icon" src="https://openweathermap.org/img/wn/${icon}.png" alt="${desc}">

            <div class="h-temp">${temp}°</div>

            <div class="h-desc">${desc}</div>

          `;

          hourly.appendChild(box);

        });


        renderChart(labels, temps);

        msg.textContent = "";

      } catch (error) {

        console.error(error);

        msg.textContent = "發生錯誤，請稍後再試。";

      }

    }


    function renderChart(labels, temps) {

      const ctx = document.getElementById("tempChart").getContext("2d");


      if (weatherChart) {

        weatherChart.destroy();

      }


      weatherChart = new Chart(ctx, {

        type: "line",

        data: {

          labels: labels,

          datasets: [

            {

              label: "溫度 (°C)",

              data: temps,

              borderWidth: 3,

              tension: 0.35,

              fill: true,

              pointRadius: 5,

              pointHoverRadius: 6,

              backgroundColor: "rgba(255,255,255,0.20)",

              borderColor: "rgba(255,255,255,0.95)",

              pointBackgroundColor: "rgba(255,255,255,1)",

              pointBorderColor: "rgba(255,255,255,1)"

            }

          ]

        },

        options: {

          responsive: true,

          maintainAspectRatio: true,

          plugins: {

            legend: {

              labels: {

                color: "#ffffff",

                font: {

                  size: 14

                }

              }

            }

          },

          scales: {

            x: {

              ticks: {

                color: "#ffffff"

              },

              grid: {

                color: "rgba(255,255,255,0.15)"

              }

            },

            y: {

              ticks: {

                color: "#ffffff"

              },

              grid: {

                color: "rgba(255,255,255,0.15)"

              }

            }

          }

        }

      });

    }


    document.getElementById("cityInput").addEventListener("keydown", function (e) {

      if (e.key === "Enter") {

        searchWeather();

      }

    });


    document.getElementById("cityInput").value = "台北";

    searchWeather();

  </script>
